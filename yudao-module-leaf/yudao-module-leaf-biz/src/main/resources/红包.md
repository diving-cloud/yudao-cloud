## 1. 项目概述

### 背景介绍

联合红包项目是一个高并发系统，允许主播在运营后台创建红包，设置一定的金额和涨粉目标。红包以挂件形式出现在直播间，观众点击并关注主播后可参与抢红包活动。

### 业务价值

- **主播视角**：增加粉丝量，提升直播间活跃度
- **用户视角**：获得奖励，提升参与感
- **平台视角**：提高整体付费率，促进用户留存

### 核心流程

1. **红包创建**：运营后台创建红包，设定金额和涨粉目标
2. **红包拆分**：系统将大红包拆分为多个小红包分片
3. **红包投放**：将红包推送至目标直播间
4. **用户抢红包**：用户请求token后参与抢红包
5. **结算发放**：异步结算并发放红包金额

### 3.3 红包创建实现

```java
@Service
@Slf4j
public class RedPacketServiceImpl implements RedPacketService {

    @Autowired
    private RedPacketRepository redPacketRepository;
    
    @Autowired
    private RedPacketShardRepository redPacketShardRepository;
    
    @Autowired
    private RedisTemplate<String, Object> redisTemplate;
    
    @Override
    @Transactional
    public RedPacket createRedPacket(RedPacketCreateRequest request) {
        // 创建红包主体
        RedPacket redPacket = new RedPacket();
        redPacket.setCreatorId(request.getCreatorId());
        redPacket.setTotalAmount(request.getTotalAmount());
        redPacket.setTotalCount(request.getTotalCount());
        redPacket.setRemainingCount(request.getTotalCount());
        redPacket.setFollowersTarget(request.getFollowersTarget());
        redPacket.setCurrentFollowers(0);
        redPacket.setStatus(RedPacketStatus.NOT_STARTED);
        redPacket.setStartTime(request.getStartTime());
        redPacket.setEndTime(request.getEndTime());
        redPacket.setCreateTime(LocalDateTime.now());
        redPacket.setUpdateTime(LocalDateTime.now());
        
        RedPacket savedRedPacket = redPacketRepository.save(redPacket);
        
        // 保存红包分片
        for (RedPacketShardDTO shardDTO : request.getShards()) {
            RedPacketShard shard = new RedPacketShard();
            shard.setRedPacketId(savedRedPacket.getId());
            shard.setShardId(shardDTO.getShardId());
            shard.setAmount(shardDTO.getAmount());
            shard.setStatus(RedPacketShardStatus.NOT_GRABBED);
            shard.setCreateTime(LocalDateTime.now());
            shard.setUpdateTime(LocalDateTime.now());
            redPacketShardRepository.save(shard);
            
            // 将红包分片数据存入Redis
            String redisKey = "red_packet:" + savedRedPacket.getId() + ":shard:" + shard.getShardId();
            redisTemplate.opsForValue().set(redisKey, shard.getAmount().toString(), 24, TimeUnit.HOURS);
        }
        
        return savedRedPacket;
    }
}
```

### 3.4 Token生成实现

```java
@Override
public RedPacketTokenDTO generateToken(Long redPacketId, Long userId) {
    // 查询红包信息
    RedPacket redPacket = redPacketRepository.findById(redPacketId)
        .orElseThrow(() -> new RuntimeException("红包不存在"));
        
    // 检查红包状态
    if (redPacket.getStatus() != RedPacketStatus.IN_PROGRESS) {
        throw new RuntimeException("红包不在进行中状态");
    }
    
    // 检查用户是否已经抢过红包
    Optional<RedPacketRecord> existingRecord = redPacketRecordRepository.findByRedPacketIdAndUserId(redPacketId, userId);
    if (existingRecord.isPresent()) {
        throw new RuntimeException("您已经抢过该红包了");
    }
    
    // 生成token
    RedPacketTokenDTO tokenDTO = new RedPacketTokenDTO();
    tokenDTO.setRedPacketId(redPacketId);
    tokenDTO.setUserId(userId);
    tokenDTO.setTimestamp(System.currentTimeMillis());
    
    // 根据用户等级、活跃度等设置金额范围
    tokenDTO.setMinAmount(new BigDecimal("0.01"));
    tokenDTO.setMaxAmount(new BigDecimal("10.00"));
    
    // 根据剩余红包数量决定是否允许抢红包
    tokenDTO.setCanGrab(redPacket.getRemainingCount() > 0);
    
    // 生成签名
    String signStr = redPacketId + ":" + userId + ":" + tokenDTO.getTimestamp() + ":" + TOKEN_SECRET;
    tokenDTO.setSign(DigestUtils.md5DigestAsHex(signStr.getBytes()));
    
    // 将token存入Redis
    String redisKey = "red_packet_token:" + redPacketId + ":" + userId;
    redisTemplate.opsForValue().set(redisKey, tokenDTO, TOKEN_EXPIRE, TimeUnit.SECONDS);
    
    return tokenDTO;
}
```

### 3.5 抢红包实现

```java
@Override
@Transactional
public BigDecimal grabRedPacket(RedPacketTokenDTO tokenDTO) {
    // 验证token
    String redisKey = "red_packet_token:" + tokenDTO.getRedPacketId() + ":" + tokenDTO.getUserId();
    RedPacketTokenDTO storedToken = (RedPacketTokenDTO) redisTemplate.opsForValue().get(redisKey);
    
    if (storedToken == null) {
        throw new RuntimeException("Token已过期");
    }
    
    // 验证签名
    String signStr = tokenDTO.getRedPacketId() + ":" + tokenDTO.getUserId() + ":" + tokenDTO.getTimestamp() + ":" + TOKEN_SECRET;
    String expectedSign = DigestUtils.md5DigestAsHex(signStr.getBytes());
    if (!expectedSign.equals(tokenDTO.getSign())) {
        throw new RuntimeException("无效的token签名");
    }
    
    // 检查是否可以抢红包
    if (!tokenDTO.getCanGrab()) {
        throw new RuntimeException("不可抢红包");
    }
    
    // 使用分布式锁防止一个用户并发抢多个红包
    String lockKey = "red_packet_lock:" + tokenDTO.getRedPacketId() + ":" + tokenDTO.getUserId();
    Boolean locked = redisTemplate.opsForValue().setIfAbsent(lockKey, "1", 10, TimeUnit.SECONDS);
    
    if (locked == null || !locked) {
        throw new RuntimeException("操作过于频繁，请稍后重试");
    }
    
    try {
        // 获取红包信息
        RedPacket redPacket = redPacketRepository.findById(tokenDTO.getRedPacketId())
            .orElseThrow(() -> new RuntimeException("红包不存在"));
        
        // 检查红包状态
        if (redPacket.getStatus() != RedPacketStatus.IN_PROGRESS) {
            throw new RuntimeException("红包不在进行中状态");
        }
        
        // 检查红包是否已抢光
        if (redPacket.getRemainingCount() <= 0) {
            throw new RuntimeException("红包已抢光");
        }
        
        // 检查用户是否已经抢过红包
        Optional<RedPacketRecord> existingRecord = redPacketRecordRepository.findByRedPacketIdAndUserId(
                tokenDTO.getRedPacketId(), tokenDTO.getUserId());
        if (existingRecord.isPresent()) {
            throw new RuntimeException("您已经抢过该红包了");
        }
        
        // 获取可用的分片
        Integer shardId = getAvailableShardId(tokenDTO.getRedPacketId());
        if (shardId == null) {
            throw new RuntimeException("红包已抢光");
        }
        
        // 获取分片金额
        String shardKey = "red_packet:" + tokenDTO.getRedPacketId() + ":shard:" + shardId;
        String amountStr = (String) redisTemplate.opsForValue().get(shardKey);
        
        if (amountStr == null) {
            throw new RuntimeException("红包分片不存在");
        }
        
        BigDecimal amount = new BigDecimal(amountStr);
        
        // 删除Redis中的分片，防止重复抢
        redisTemplate.delete(shardKey);
        
        // 更新红包主体信息
        redPacket.setRemainingCount(redPacket.getRemainingCount() - 1);
        if (redPacket.getRemainingCount() == 0) {
            redPacket.setStatus(RedPacketStatus.FINISHED);
        }
        redPacketRepository.save(redPacket);
        
        // 创建抢红包记录
        RedPacketRecord record = new RedPacketRecord();
        record.setRedPacketId(tokenDTO.getRedPacketId());
        record.setUserId(tokenDTO.getUserId());
        record.setShardId(shardId);
        record.setAmount(amount);
        record.setStatus(RedPacketRecordStatus.PENDING);
        record.setCreateTime(LocalDateTime.now());
        record.setUpdateTime(LocalDateTime.now());
        redPacketRecordRepository.save(record);
        
        // 发送MQ消息，异步处理结算
        rocketMQTemplate.convertAndSend("red-packet-settlement", record);
        
        return amount;
    } finally {
        // 释放锁
        redisTemplate.delete(lockKey);
    }
}

// 获取可用分片ID
private Integer getAvailableShardId(Long redPacketId) {
    // 从Redis获取可用分片
    String redisKey = "red_packet:available_shards:" + redPacketId;
    Set<ZSetOperations.TypedTuple<Object>> shardWithCounts = 
        redisTemplate.opsForZSet().reverseRangeWithScores(redisKey, 0, 0);
    
    if (shardWithCounts != null && !shardWithCounts.isEmpty()) {
        ZSetOperations.TypedTuple<Object> tuple = shardWithCounts.iterator().next();
        Integer shardId = Integer.valueOf(tuple.getValue().toString());
        Double remainingCount = tuple.getScore();
        
        // 如果该分片还有剩余，则分配该分片
        if (remainingCount > 0) {
            // 原子性递减该分片剩余数量
            redisTemplate.opsForZSet().incrementScore(redisKey, shardId, -1);
            return shardId;
        }
    }
    
    return null;
}
```

### 3.6 结算消费者实现

```java
@Service
@RocketMQMessageListener(topic = "red-packet-settlement", consumerGroup = "red-packet-consumer")
@Slf4j
public class RedPacketSettlementConsumer implements RocketMQListener<RedPacketRecord> {

    @Autowired
    private RedPacketRecordRepository redPacketRecordRepository;
    
    @Autowired
    private PaymentService paymentService;

    @Override
    @Transactional
    public void onMessage(RedPacketRecord record) {
        try {
            // 查询最新的记录状态，防止重复结算
            RedPacketRecord latestRecord = redPacketRecordRepository.findById(record.getId())
                .orElseThrow(() -> new RuntimeException("记录不存在"));
            
            if (latestRecord.getStatus() != RedPacketRecordStatus.PENDING) {
                log.info("红包记录 {} 状态为 {}，跳过结算", latestRecord.getId(), latestRecord.getStatus());
                return;
            }
            
            // 执行支付逻辑
            boolean success = paymentService.transferMoney(record.getUserId(), record.getAmount());
            
            // 更新结算状态
            latestRecord.setStatus(success ? RedPacketRecordStatus.SETTLED : RedPacketRecordStatus.FAILED);
            latestRecord.setUpdateTime(LocalDateTime.now());
            redPacketRecordRepository.save(latestRecord);
            
            log.info("红包记录 {} 结算完成，状态：{}", latestRecord.getId(), latestRecord.getStatus());
        } catch (Exception e) {
            log.error("红包结算失败: " + e.getMessage(), e);
        }
    }
}
```

## 4. 高频面试问题与解答

### 4.1 如何设计这个系统来应对高并发场景？

**答案**：
设计高并发红包系统需要多层次优化：

1. **前端限流**：通过Token预请求机制控制总流量，将可参与用户数量限制在可控范围内。

2. **架构分层**：使用微服务架构将创建红包、抢红包功能分离，独立扩容。

3. **无状态设计**：服务无状态化，便于水平扩展。

4. **缓存策略**：
    - Redis预加载所有红包分片数据
    - 本地缓存热点数据
    - 多级缓存设计减轻DB压力

5. **异步处理**：抢红包与结算完全分离，使用RocketMQ异步处理转账逻辑。

6. **批量处理**：结算环节采用批量处理方式，提高吞吐量。

7. **热点分散**：红包分片策略避免单点热数据，将访问压力分散到多个Redis实例。

### 4.2 红包系统中如何处理分布式锁来避免超发问题？

**答案**：
分布式锁是解决红包超发的关键：

1. **锁选型**：使用Redis的`SETNX`命令实现分布式锁，因其高性能和可靠性。

2. **锁粒度设计**：
    - 对每个用户抢特定红包设置锁 `red_packet_lock:{redPacketId}:{userId}`
    - 避免全局锁，精细化锁粒度提高并发性

3. **防死锁机制**：设置合理的锁超时时间(如10秒)，防止持有者崩溃导致死锁。

4. **锁释放保证**：使用`try-finally`模式确保锁总能释放。

5. **锁失败处理**：获取锁失败时返回友好提示"操作频繁，请稍后重试"，而非阻塞等待。

6. **双重检查**：获取锁后再次检查库存和用户是否已抢，作为二次保障。

7. **可重入考虑**：在复杂业务逻辑中考虑锁的可重入性。

### 4.3 如何优化Redis在红包场景下的性能？

**答案**：
Redis性能优化是红包系统成功的关键：

1. **数据结构选择**：
    - 使用String类型存储红包分片金额，而非Hash
    - 原因：原子操作性能更好，单key访问速度更快

2. **键设计策略**：
    - 采用`red_packet:{redPacketId}:shard:{shardId}`格式
    - 便于批量操作和过期管理

3. **内存优化**：
    - 设置合理的过期时间(24小时)避免内存泄漏
    - 使用压缩编码减少内存占用

4. **集群设计**：
    - Redis集群部署，将不同红包数据分散到不同节点
    - 使用一致性哈希算法减少集群扩容时的数据迁移

5. **连接池管理**：
    - 合理配置连接池大小和超时时间
    - 监控连接池使用情况，动态调整

6. **批量操作**：使用pipeline批量获取中奖记录，减少网络往返。

7. **避免大key**：红包记录过多时进行分片存储，避免单个key过大。

8. **读写分离**：对于获奖记录查询等读多写少场景，采用读写分离。

### 4.4 如何保证红包业务的资金安全？

**答案**：
资金安全是红包系统的生命线：

1. **预分配机制**：
    - 红包创建时即完成金额分配和拆分
    - 严格控制总金额与分片金额之和完全一致

2. **库存隔离**：
    - Redis控制可用库存，DB保存最终记录
    - 双重校验库存合法性

3. **幂等性设计**：
    - 每个用户对每个红包生成唯一标识
    - 数据库唯一索引约束(redPacketId+userId)
    - 确保一个用户只能抢一次红包

4. **事务补偿机制**：
    - 使用RocketMQ事务消息确保转账操作的原子性
    - 失败后自动重试或人工介入

5. **对账系统**：
    - 定时任务比对红包总额与已抢金额之和
    - 比对红包记录与实际转账记录

6. **操作日志**：
    - 记录所有关键操作，支持审计和回溯
    - 异常情况下能还原完整操作轨迹

7. **风控机制**：
    - Token生成阶段进行风险控制
    - 对异常用户行为实时监控和拦截

### 4.5 RocketMQ在项目中的作用以及如何保证消息可靠性？

**答案**：
RocketMQ在红包系统中扮演着关键角色：

### 4.7 如何解决缓存与数据库的一致性问题？

   ```java
   redisTemplate.delete(key);  // 再次删除
   ```

3. **消息队列保证最终一致性**：
    - 将数据变更通过消息队列异步通知缓存更新
    - 即使部分失败，重试机制也能保证最终一致

4. **读写锁**：
    - 对热点数据加读写锁
    - 写操作时阻塞读，保证一致性

5. **版本号机制**：
    - 缓存数据添加版本号
    - 读取时比对版本，不一致则回源

6. **定时同步任务**：
    - 定期全量同步数据库到缓存
    - 作为最终保障机制

7. **监控不一致**：
    - 抽样检测缓存与数据库一致性
    - 发现不一致时告警并修复

8. **分布式锁保护**：
    - 更新操作加分布式锁
    - 防止并发更新导致的不一致

### 4.8 如何防止缓存穿透、击穿和雪崩问题？

**答案**：
缓存问题防护对高并发系统至关重要：

1. **缓存穿透防护**：
    - **定义**：查询不存在的数据，每次都击穿到数据库
    - **解决方案**：
        - 对空结果也进行缓存，设置较短过期时间
        - 使用布隆过滤器快速判断key是否存在
        - Redis实现布隆过滤器示例：
      ```java
      // 使用Redis bitmap实现简易布隆过滤器
      redisTemplate.opsForValue().setBit("bf:redpackets", hashFunction(redPacketId), true);
      // 检查是否可能存在
      Boolean exists = redisTemplate.opsForValue().getBit("bf:redpackets", hashFunction(redPacketId));
      ```

2. **缓存击穿防护**：
    - **定义**：热点key过期瞬间被大量请求击穿到数据库
    - **解决方案**：
        - 互斥锁机制：首个请求获取锁并更新缓存，其他请求等待
        - 热点数据永不过期或提前异步更新
      ```java
      // 伪代码：缓存击穿保护
      String lockKey = "lock:" + redPacketId;
      if (redisTemplate.opsForValue().setIfAbsent(lockKey, "1", 10, TimeUnit.SECONDS)) {
          try {
              // 从数据库加载并更新缓存
          } finally {
              redisTemplate.delete(lockKey);
          }
      } else {
          // 短暂睡眠后重试从缓存获取
          Thread.sleep(50);
          return getFromCache(redPacketId);
      }
      ```

3. **缓存雪崩防护**：
    - **定义**：大量缓存同时过期，导致数据库压力激增
    - **解决方案**：
        - 过期时间添加随机值，避免同时过期
        - 多级缓存架构：本地缓存+分布式缓存
        - 服务熔断和降级机制
        - 提前演练和扩容机制

4. **系统级防护措施**：
    - 对缓存实例进行监控，及时发现异常
    - 建立缓存预热机制，避免冷启动问题
    - 关键业务添加降级开关，极端情况下可快速降级

### 4.9 如何设计红包金额拆分算法，保证公平性和随机性？

**答案**：
红包拆分算法需兼顾公平性、随机性和趣味性：

1. **二倍均值法**：
    - 核心思想：每次随机金额的上限是剩余均值的2倍
    - 优点：保证每人都能抢到金额，且金额分布较为均衡
    - 实现示例：
   ```java
   public List<BigDecimal> splitRedPacket(BigDecimal totalAmount, int totalCount) {
       List<BigDecimal> result = new ArrayList<>(totalCount);
       BigDecimal remainingAmount = totalAmount;
       int remainingCount = totalCount;
       
       Random random = new Random();
       for (int i = 0; i < totalCount - 1; i++) {
           // 计算本次可领取的最大金额 = 剩余平均值 * 2
           BigDecimal maxAmount = remainingAmount.divide(new BigDecimal(remainingCount), 2, RoundingMode.DOWN)
               .multiply(new BigDecimal(2));
           
           // 随机抽取金额
           int maxAmountInt = maxAmount.multiply(new BigDecimal(100)).intValue();
           int amount = random.nextInt(maxAmountInt) + 1; // +1保证最少1分
           BigDecimal redPacketAmount = new BigDecimal(amount).divide(new BigDecimal(100), 2, RoundingMode.DOWN);
           
           result.add(redPacketAmount);
           remainingAmount = remainingAmount.subtract(redPacketAmount);
           remainingCount--;
       }
       
       // 最后一个红包直接把余额给他
       result.add(remainingAmount);
       return result;
   }
   ```

2. **线段切割法**：
    - 将总金额看作线段，随机切N-1刀，形成N个片段
    - 实现更均匀的随机分布

3. **约束条件**：
    - 每个红包至少有1分钱
    - 最大红包不超过总金额的一定比例（如30%）

4. **改进策略**：
    - 添加权重因子，根据用户活跃度调整概率
    - 引入波动系数，控制红包金额的离散程度

5. **数据验证**：
    - 严格确保红包总和等于预设金额
    - 每个分片金额满足系统设置的最小值要求

6. **安全考虑**：
    - 服务端生成拆分金额，避免客户端作弊
    - 添加随机因子，防止金额规律被识破

### 4.13 如何保证每一个红包分片都会被抢完？

3. **分片动态分配策略**：

   ```java
   @Service
   public class DynamicShardingService {
       @Autowired
       private RedisTemplate<String, Object> redisTemplate;
       
       // 获取可用分片ID
       public Integer getAvailableShardId(Long redPacketId, Long userId) {
           String redisKey = "red_packet:available_shards:" + redPacketId;
           
           // 1. 优先尝试使用有序集合中权重最高的分片（剩余数量最多的）
           Set<ZSetOperations.TypedTuple<Object>> shardWithCounts = 
               redisTemplate.opsForZSet().reverseRangeWithScores(redisKey, 0, 0);
           
           if (shardWithCounts != null && !shardWithCounts.isEmpty()) {
               ZSetOperations.TypedTuple<Object> tuple = shardWithCounts.iterator().next();
               Integer shardId = Integer.valueOf(tuple.getValue().toString());
               Double remainingCount = tuple.getScore();
               
               // 如果该分片还有剩余，则分配该分片
               if (remainingCount > 0) {
                   // 原子性递减该分片剩余数量
                   redisTemplate.opsForZSet().incrementScore(redisKey, shardId, -1);
                   return shardId;
               }
           }
           
           // 2. 所有分片都已分配完，返回随机分片（用户可能抢不到，但至少有机会）
           List<Object> allShards = redisTemplate.opsForZSet().range(redisKey, 0, -1)
               .stream().collect(Collectors.toList());
           
           if (allShards.isEmpty()) {
               return null; // 所有分片都已被抢光
           }
           
           return Integer.valueOf(allShards.get(new Random().nextInt(allShards.size())).toString());
       }
   }
   ```

   ```java
       }

   ```

   ```java
   @Transactional
   ```

### 4.15 你们分库分表是怎么做的？

**答案**：
我们的分库分表策略是基于业务增长预测和实际性能测试精心设计的：

1. **分片维度选择**：
    - **红包主表**：按照红包ID哈希分片，确保单个红包的基本信息在同一个分片
    - **红包分片表**：也按红包ID分片，与主表保持一致，便于联合查询
    - **红包记录表**：采用红包ID+用户ID的组合分片，更细粒度分散读写压力

2. **中间件选型与实现**：
    - 主要采用ShardingSphere作为分库分表中间件
    - 关键配置示例：
   ```yaml
   # ShardingSphere配置示例
   shardingsphere:
     datasource:
       names: ds0,ds1,ds2,ds3
       # ... 数据源配置 ...
     rules:
       sharding:
         tables:
           red_packet:
             actual-data-nodes: ds$->{0..3}.red_packet_$->{0..7}
             database-strategy:
               standard:
                 sharding-column: id
                 sharding-algorithm-name: red-packet-db-algorithm
             table-strategy:
               standard:
                 sharding-column: id
                 sharding-algorithm-name: red-packet-tbl-algorithm
           red_packet_record:
             actual-data-nodes: ds$->{0..3}.red_packet_record_$->{0..15}
             database-strategy:
               complex:
                 sharding-columns: red_packet_id,user_id
                 sharding-algorithm-name: record-complex-db-algorithm
             table-strategy:
               complex:
                 sharding-columns: red_packet_id,user_id
                 sharding-algorithm-name: record-complex-tbl-algorithm
   ```

3. **分片算法实现**：
   ```java
   public class RedPacketRecordComplexShardingAlgorithm implements ComplexKeysShardingAlgorithm<Long> {
       @Override
       public Collection<String> doSharding(Collection<String> availableTargetNames, 
                                          ComplexKeysShardingValue<Long> shardingValue) {
           Map<String, Collection<Long>> columnNameAndShardingValuesMap = shardingValue.getColumnNameAndShardingValuesMap();
           Collection<Long> redPacketIds = columnNameAndShardingValuesMap.get("red_packet_id");
           Collection<Long> userIds = columnNameAndShardingValuesMap.get("user_id");
           
           // 优先使用红包ID分片，兼顾用户ID分散读写压力
           List<String> result = new ArrayList<>();
           for (Long redPacketId : redPacketIds) {
               for (Long userId : userIds) {
                   // 组合哈希算法，降低冲突概率
                   int shardingKey = (redPacketId.hashCode() * 31 + userId.hashCode()) % 16;
                   if (shardingKey < 0) shardingKey += 16;
                   
                   for (String targetName : availableTargetNames) {
                       if (targetName.endsWith("_" + shardingKey)) {
                           result.add(targetName);
                       }
                   }
               }
           }
           return result;
       }
   }
   ```

4. **扩容策略**：
    - 使用双写+追数据方案平滑迁移
    - 扩容时采用一致性哈希减少数据迁移量

5. **分布式事务处理**：
    - 跨片事务采用Best Efforts 1PC模式提高性能
    - 关键业务使用TCC模式保证最终一致性

6. **查询优化**：
    - 针对跨分片查询进行SQL重写和优化
    - 建立合适的二级索引减少跨分片查询

7. **监控与运维**：
    - 建立分片健康度监控，包括分片间数据平衡度、查询性能等
    - 慢SQL实时告警并自动提供优化建议
    - 定期执行再平衡任务，保持各分片负载均衡

这套分库分表方案在我们的红包系统中经受住了多次大促活动的考验，支持了从每日百万级到千万级的业务增长。

### 4.16 你们的token加密是怎么做的？

**答案**：
我们的token加密方案融合了多种安全技术，确保抢红包过程的安全性：

1. **多层加密架构**：
    - **数据加密层**：敏感信息AES加密
    - **签名验证层**：整体数据HMAC-SHA256签名
    - **传输安全层**：使用TLS 1.3加密传输

2. **Token内容设计**：
   ```json
   {
     "payload": {
       "rpid": 12345678,       // 红包ID
       "uid": 87654321,        // 用户ID
       "exp": 1634567890,      // 过期时间戳
       "min": 0.01,            // 最小金额
       "max": 10.00,           // 最大金额
       "cgb": true,            // 是否可抢
       "nce": "abcd1234"       // 随机数防重放
     },
     "sign": "计算得到的签名"
   }
   ```

3. **签名生成算法**：
   ```java
   public String generateToken(RedPacketTokenDTO tokenDTO) {
       // 1. 构建原始数据
       Map<String, Object> payload = new HashMap<>();
       payload.put("rpid", tokenDTO.getRedPacketId());
       payload.put("uid", tokenDTO.getUserId());
       payload.put("exp", System.currentTimeMillis() + TOKEN_EXPIRE * 1000);
       payload.put("min", tokenDTO.getMinAmount());
       payload.put("max", tokenDTO.getMaxAmount());
       payload.put("cgb", tokenDTO.getCanGrab());
       payload.put("nce", generateNonce());  // 生成随机数
       
       // 2. 将 payload 转为 JSON
       String jsonPayload = objectMapper.writeValueAsString(payload);
       
       // 3. AES加密payload
       String encryptedPayload = aesEncrypt(jsonPayload, AES_SECRET_KEY);
       
       // 4. 计算HMAC签名
       String dataToSign = encryptedPayload + "." + tokenDTO.getUserId() + "." + TOKEN_VERSION;
       String signature = calculateHMAC(dataToSign, HMAC_SECRET_KEY);
       
       // 5. 组合最终token
       Map<String, String> tokenMap = new HashMap<>();
       tokenMap.put("payload", encryptedPayload);
       tokenMap.put("sign", signature);
       
       // 6. Base64编码整体token
       return Base64.getUrlEncoder().encodeToString(objectMapper.writeValueAsString(tokenMap).getBytes());
   }
   ```

4. **密钥管理机制**：
    - 使用HSM（硬件安全模块）存储根密钥
    - 应用层密钥定期轮转（每周一次）
    - 多环境使用不同密钥，开发测试与生产完全隔离

5. **安全增强措施**：

    - **设备绑定**：token可选与设备指纹绑定

### 4.17 你们开发上线过程中有没有遇到什么问题，是如何排查解决的？

**答案**：
在红包系统上线过程中，我们遇到了一个典型问题：大促活动中红包抢夺成功率急剧下降。

1. **问题现象**：
    - 用户反馈"手慢了，红包被抢光"，但实际系统红包库存充足
    - 错误日志显示大量"Failed to acquire Redis lock"异常
    - Redis CPU使用率飙升至90%以上

2. **排查过程**：
    - **第一步：确认问题范围**
        * 通过灰度分析确认问题只出现在特定区域的用户
        * 初步判断可能是区域性Redis集群问题

    - **第二步：监控数据分析**
        * Redis监控显示SETNX命令异常高的执行时间（从0.2ms飙升至20ms）
        * Redis慢查询日志显示大量分布式锁相关命令

    - **第三步：代码审查**
      ```java
      // 存在问题的代码
      Boolean locked = redisTemplate.opsForValue().setIfAbsent(lockKey, "1", 10, TimeUnit.SECONDS);
      if (locked == null || !locked) {
          throw new RuntimeException("操作过于频繁，请稍后重试");
      }
      ```
        * 问题点：直接抛出异常会导致用户体验差，且不会重试

    - **第四步：深入问题根因**
        * 使用Arthas分析线程栈，发现大量线程阻塞在Redis连接获取上
        * Redis集群主从复制延迟高达300ms，导致从库无法及时响应
        * 根因：Redis主从节点间网络分区，触发了故障转移，但客户端连接池未及时感知主从切换

3. **解决方案**：
    - **紧急措施**：
      ```java
      // 改进后的代码
      int retryCount = 3;
      boolean locked = false;
      
      while (retryCount > 0 && !locked) {
          try {
              locked = redisTemplate.opsForValue().setIfAbsent(lockKey, "1", 10, TimeUnit.SECONDS);
              if (locked) break;
              
              // 指数退避重试
              Thread.sleep((long) (10 * Math.pow(2, 3 - retryCount)));
              retryCount--;
          } catch (InterruptedException e) {
              Thread.currentThread().interrupt();
              break;
          } catch (Exception e) {
              log.warn("获取锁异常，尝试重试", e);
              retryCount--;
          }
      }
      
      if (!locked) {
          return ResponseEntity.status(HttpStatus.TOO_MANY_REQUESTS)
              .body("系统繁忙，请稍后再试");
      }
      ```

    - **中期措施**：
        * 升级Redis客户端，使用Lettuce替换Jedis，支持自动主从感知
        * 实施Redis读写分离，分散压力
        * 优化分布式锁实现，使用Redisson替代自定义实现

    - **长期解决方案**：
        * 重构Redis架构，从主从结构升级为Redis Cluster
        * 引入熔断机制，自动隔离不稳定节点
        * 完善监控，添加Redis主从复制延迟等关键指标

4. **效果验证**：
    - 紧急措施实施后，成功率恢复至95%以上
    - 一周后完成中期措施，成功率稳定在99%

5. **经验总结**：
    - 分布式锁需考虑网络抖动和主从切换场景
    - 异常处理策略要考虑用户体验，适当重试优于直接失败
    - Redis主从架构在网络不稳定情况下存在脆弱性
    - 完善的监控对于快速定位问题至关重要

### 4.18 你们是如何做对账的？

**答案**：
对账系统是保障红包资金安全的最后防线，我们构建了一套多层次的对账体系：

1. **对账频率与范围**：
    - **实时对账**：每笔红包结算完成即触发轻量级对账
    - **小时级对账**：每小时一次增量对账，针对近期活动
    - **日对账**：每日凌晨进行全量对账
    - **周对账**：周末进行跨系统大对账，包含支付系统、账户系统等

2. **多维度对账策略**：
    - **红包创建对账**：创建红包总金额 = 所有分片金额之和
    - **红包库存对账**：已发放 + 未发放 = 总分片数
    - **红包金额对账**：已发放总金额 + 剩余金额 = 红包总金额
    - **账务系统对账**：红包发放记录 = 账务系统转账记录

3. **对账代码实现**：
   ```java
   @Service
   @Slf4j
   public class RedPacketReconciliationService {
       
       @Autowired
       private RedPacketRepository redPacketRepository;
       
       @Autowired
       private RedPacketShardRepository shardRepository;

           LocalDate yesterday = LocalDate.now().minusDays(1);
           LocalDateTime startTime = yesterday.atStartOfDay();
           LocalDateTime endTime = yesterday.plusDays(1).atStartOfDay();

           reconcileWithPaymentSystem(startTime, endTime);
       }
       
       private void reconcileWithPaymentSystem(LocalDateTime startTime, LocalDateTime endTime) {
           // 获取已结算的红包记录
           List<RedPacketRecord> settledRecords = recordRepository.findByStatusAndUpdateTimeBetween(
                   RedPacketRecordStatus.SETTLED, startTime, endTime);
           
           // 批量获取支付系统转账记录
           List<String> businessIds = settledRecords.stream()
                   .map(r -> "RP_" + r.getId())
                   .collect(Collectors.toList());
           
           Map<String, PaymentRecord> paymentRecords = paymentClient.batchQueryByBusinessIds(businessIds);
           
           // 逐条对账
           for (RedPacketRecord record : settledRecords) {
               String businessId = "RP_" + record.getId();
               PaymentRecord paymentRecord = paymentRecords.get(businessId);
               
               if (paymentRecord == null) {
                   // 支付系统无记录
                   log.error("支付系统缺失转账记录: 红包记录ID={}", record.getId());
                   createMissingPaymentRecord(record);
                   continue;
               }
               
               // 金额对账
               if (record.getAmount().compareTo(paymentRecord.getAmount()) != 0) {
                   log.error("红包金额与支付系统不一致: 红包记录ID={}, 红包金额={}, 支付系统金额={}",
                           record.getId(), record.getAmount(), paymentRecord.getAmount());
                   
                   createDiscrepancyRecord(record.getId(), "PAYMENT_AMOUNT", 
                           record.getAmount(), paymentRecord.getAmount());
               }
           }
       }
       
       // 其他对账方法实现...
   }
   ```

4. **差异处理机制**：
    - **自动修复**：对于明确原因的小额差异，系统自动修复
    - **人工审核**：大额差异进入工单系统，由财务人员审核
    - **处理流程**：差异分析 → 制定修复方案 → 执行修复 → 再次对账验证

    - **数据一致性保障**：使用消息队列确保对账数据完整采集
    - **高性能设计**：大对账任务采用分片并行处理
    - **横向可扩展**：支持按业务线或时间维度扩展对账能力

6. **安全与审计**：
    - 对账配置和规则变更需多人
